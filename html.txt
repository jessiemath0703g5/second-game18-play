        function shuffleArray(array) {
            let arr = array.slice();
            for (let i = arr.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [arr[i], arr[j]] = [arr[j], arr[i]];
            }
            return arr;
        }

        function generateQuestion(type) {
            const niceTotals = [10, 20, 25, 40, 50, 100, 120, 150, 200, 250, 300, 400, 500, 800, 1000];
            const gridTotals = [10, 20, 25, 40, 50, 100];
            
            if (type === 1) {
                let T = rand(niceTotals);
                let targetP = rand([10, 15, 20, 25, 30, 35, 40, 45, 50, 60, 70, 75, 80, 85, 90]);
                let P = Math.round(T * targetP / 100);
                if (P === 0) P = 1; 
                if (P >= T) P = T - 1;
                let ans = Math.round((P / T) * 100); 
                let N = T - P;
                return { T, P, N, ans, trap: rand(['normal', 'complement']), theme: getSessionTheme('t1', themesT1) };
            }
            else if (type === 2) {
                let T = rand(gridTotals);
                let C = Math.floor(Math.random() * (T - 1)) + 1;
                let ansPercent = Math.round((C / T) * 100);
                return { T, C, ansPercent, theme: getSessionTheme('t2', themesT2) };
            }
            else if (type === 3) {
                let T = rand(niceTotals);
                let percBase = Math.floor(Math.random() * 19 + 1) * 5; 
                let P = Math.round((T * percBase) / 100);
                if(P === 0 || P >= T) { P = Math.floor(T/2); }
                let I = T - P;
                let g = gcd(P, T);
                return { 
                    T, p: P, i: I, 
                    num: P / g, den: T / g, 
                    perc: Math.round((P / T) * 100),
                    trap: rand(['normal', 'totalGiven']), theme: getSessionTheme('t3', themesT3)
                };
            }
            else if (type === 4) {
                let arr = [];
                for(let j=0; j<5; j++) {
                    let qType = rand(['P2D', 'D2P', 'F2P', 'P2F', 'D2F', 'F2D']);
                    let dens = [2, 4, 5, 8, 10, 20, 25, 40, 50, 100, 3, 6, 7];
                    let den = rand(dens);
                    let num = Math.floor(Math.random()*(den-1)) + 1;
                    let _g = gcd(num, den);
                    num /= _g; den /= _g;

                    let isExact = (10000 % den === 0);
                    let dec, decStr, percStr;

                    if (isExact) {
                        dec = num / den;
                        decStr = String(Number(dec.toFixed(4)));
                        percStr = String(Number((dec * 100).toFixed(2)));
                    } else {
                        dec = Math.round((num / den) * 100) / 100;
                        decStr = String(dec);
                        percStr = String(Math.round((num / den) * 100));
                    }

                    let needsRounding = !isExact;
                    let modeTitle = "", qHtml = "", ansHtml = "", reqFrac = false;
                    let ansVal = null, ansNum = null, ansDen = null;
                    let formatHtmlFrac = (n, d) => `<span class="frac-inline"><span>${n}</span><span>${d}</span></span>`;

                    switch(qType) {
                        case 'P2D': modeTitle = "百分率 轉換為 小數"; qHtml = `${percStr}%`; ansHtml = decStr; ansVal = parseFloat(decStr); break;
                        case 'D2P': modeTitle = "小數 轉換為 百分率"; qHtml = decStr; ansHtml = `${percStr}%`; ansVal = parseFloat(percStr); break;
                        case 'F2P': modeTitle = "分數 轉換為 百分率"; qHtml = formatHtmlFrac(num, den); ansHtml = `${percStr}%`; ansVal = parseFloat(percStr); break;
                        case 'P2F': modeTitle = "百分率 轉換為 分數"; qHtml = `${percStr}%`; ansHtml = formatHtmlFrac(num, den); reqFrac = true; ansNum = num; ansDen = den; break;
                        case 'D2F': modeTitle = "小數 轉換為 分數"; qHtml = decStr; ansHtml = formatHtmlFrac(num, den); reqFrac = true; ansNum = num; ansDen = den; break;
                        case 'F2D': modeTitle = "分數 轉換為 小數"; qHtml = formatHtmlFrac(num, den); ansHtml = decStr; ansVal = parseFloat(decStr); break;
                    }
                    arr.push({ type: qType, modeTitle, qHtml, ansHtml, reqFrac, ansNum, ansDen, ansVal, needsRounding });
                }
                return { arr };
            }
            else if (type === 5) {
                let zonesCount = rand([3, 4]);
                let zones = [];
                for(let z=0; z<zonesCount; z++){
                    let T = rand(niceTotals);
                    let a = Math.floor(Math.random() * 15 + 5) * 5;
                    let n = Math.round((T * a) / 100);
                    a = Math.round((n / T) * 100);
                    zones.push({ name: String.fromCharCode(65+z), n: n, t: T, a: a });
                }
                let maxA = Math.max(...zones.map(z => z.a));
                zones.forEach(z => z.isMax = (z.a === maxA));
                return { zones: zones, theme: getSessionTheme('t5', themesT5) };
            }
        }

            document.getElementById(screenId).classList.add('active');
            document.getElementById('gameControls').style.display = screenId === 'screen-game' ? 'flex' : 'none';

        function startGame() {
            playerName = document.getElementById('playerName').value.trim();
            if (!playerName) { 
                triggerPanelEffect('error-shake', 500); // 觸發震動特效
                alert("系統警告：必須輸入特工代號！"); 
                return; 
            }
            
            // 嘗試播放背景音樂
            const bgm = document.getElementById('bgMusic');
            if (soundEnabled) {
                bgm.play().catch(e => console.log('BGM Playback prevented by browser'));
            }

            score = 0;
            currentLevel = 1;
            document.getElementById('correctIconsDisplay').innerHTML = ""; 
            
            let group1 = shuffleArray([1, 2, 3, 4, 5]);
            let group2 = shuffleArray([1, 2, 3, 4, 5]);
            gameConfig.sequence = group1.concat(group2);
            
            // 初始化本次挑戰的主題洗牌庫以確保不重複，用完才重洗
            window.sessionThemes = {
                t1: shuffleArray([...themesT1]),
                t2: shuffleArray([...themesT2]),
                t3: shuffleArray([...themesT3]),
                t5: shuffleArray([...themesT5])
            };

            switchScreen('screen-game');
            startTimer();
            loadLevel();
        }

            if(confirm("確定要放棄當前進度重來嗎？")) { startGame(); }
        }

        const canvas = document.getElementById('whiteboard-canvas');
        const ctx = canvas.getContext('2d');
        let drawing = false;

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        document.addEventListener('mousedown', (e) => {
            if(!isWhiteboardOn) return;
            const isInteractive = e.target.closest('input, button, select, .zone-card, .grid-cell, a, .sound-toggle, .keypad-btn, .keypad-close');
            if(isInteractive) {
                drawing = false;
                return;
            }
            drawing = true;
            ctx.beginPath();
            ctx.moveTo(e.clientX, e.clientY);
        });

        document.addEventListener('mouseup', () => { 
            drawing = false; 
            ctx.beginPath(); 
        });

        document.addEventListener('mousemove', (e) => {
            if(!drawing || !isWhiteboardOn) return;
            ctx.lineWidth = 3; 
            ctx.lineCap = "round"; 
            ctx.strokeStyle = "#00ffff";
            ctx.lineTo(e.clientX, e.clientY); 
            ctx.stroke(); 
            ctx.beginPath(); 
            ctx.moveTo(e.clientX, e.clientY);
        });

        function toggleWhiteboard() {
            isWhiteboardOn = !isWhiteboardOn;
            canvas.style.display = isWhiteboardOn ? 'block' : 'none';
            if (isWhiteboardOn) {
                document.body.classList.add('whiteboard-active');
            } else {
                document.body.classList.remove('whiteboard-active');
                ctx.clearRect(0, 0, canvas.width, canvas.height);
            }
        }

        function loadLevel() {
            if (currentLevel > 10) {
                finishGame(false);
                return;
            }

            let type = gameConfig.sequence[currentLevel - 1];
            let qData = generateQuestion(type); 
            window.currentQuestionData = qData;
            
            switch(type) {
                case 1: renderType1(qData); break;
                case 2: renderType2(qData); break;
                case 3: renderType3(qData); break;
                case 4: renderType4(qData); break;
                case 5: renderType5(qData); break;
            }
        }

            }
        }

            const el = document.getElementById('c'+i);

                el.classList.remove('active');
                window.litCount--;
            } else {
                el.classList.add('active');
                window.litCount++;
            }
            document.getElementById('l2-bar').style.width = (window.litCount / total * 100) + '%';
        }

        function checkType4(e) {
            if(e && e.type === 'keyup' && e.key !== 'Enter') return;
            
            const subQ = window.l4Questions[window.l4Current];
            let isCorrect = false;

            if (subQ.reqFrac) {
                let uNumInput = document.getElementById('l4-num-in');
                let uDenInput = document.getElementById('l4-den-in');
                if(!uNumInput || !uDenInput) return; 
                let uNum = uNumInput.value.trim();
                let uDen = uDenInput.value.trim();
                if (uNum === '' || uDen === '') return;
                
                uNumInput.disabled = true;
                uDenInput.disabled = true;
                isCorrect = (parseInt(uNum) === subQ.ansNum && parseInt(uDen) === subQ.ansDen);
            } else {
                let uInput = document.getElementById('l4-input');
                if(!uInput) return; 
                let uValStr = uInput.value.trim();
                if (uValStr === '') return;
                
                uInput.disabled = true;
                isCorrect = (parseFloat(uValStr) === subQ.ansVal);
            }

            const fbMini = document.getElementById('l4-feedback-mini');

            if(isCorrect) {
                playSound('success');
                window.l4SubScore += 20;
                fbMini.style.color = "#00ffff";
                fbMini.innerText = "精準打擊！ (+20分)";
            } else {
                triggerPanelEffect('error-shake', 500); // 讓小關卡錯誤也會震動提示
                playSound('error');
                fbMini.style.color = "#ff0055";
                fbMini.innerHTML = `錯誤！正解為: ${subQ.ansHtml}`;
            }

            let summaryLine = `Q${window.l4Current + 1}: ${subQ.qHtml} = ${subQ.ansHtml}`;
            window.l4Summary.push(summaryLine);

            window.l4Current++;
            setTimeout(() => {
                fbMini.innerText = "";
                
                if(window.l4Current >= window.l4Questions.length) {
                    if(window.l4SubScore === 100) {
                        handleLevelResult(true, true, "", 100);
                    } else {
                        let ansDump = window.l4Summary.join('<br><br>');
                        handleLevelResult(true, false, ansDump, window.l4SubScore);
                    }
                } else {
                    updateType4UI();
                }
            }, 1000);
        }

        function checkType5() {
            const q = window.currentQuestionData;
            let theme = q.theme;
            let pts = 0;
            let percentPts = 0;
            let selectionCorrect = true;
            
            let eachPercentVal = 50 / q.zones.length;
            
            q.zones.forEach((z, i) => {
                let uVal = parseInt(document.getElementById(`i${i}`).value);
                let isSelected = document.getElementById(`z${i}`).classList.contains('selected');
                
                if (uVal === z.a) {
                    percentPts += eachPercentVal;
                }
                
                if (isSelected !== z.isMax) {
                    selectionCorrect = false;
                }
            });

            pts = Math.round(percentPts) + (selectionCorrect ? 50 : 0);

            if(pts === 100) {
                handleLevelResult(true, true, "", 100);
            } else {
                let correctText = `各${theme.z_name}數值與解答：<br>`;
                q.zones.forEach(z => { correctText += `${z.name}${theme.z_name}: ${z.a}%<br>`; });
                let maxNames = q.zones.filter(z => z.isMax).map(z => z.name).join(' 與 ');
                correctText += `<br><span style="color:#ffd700;">正確選取應為：${maxNames}${theme.z_name}</span>`;
                handleLevelResult(true, false, correctText, pts);
            }
        }
